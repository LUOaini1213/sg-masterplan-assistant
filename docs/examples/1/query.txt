Equivalent SQL:
SELECT objectid, lu_desc, gpr, ST_Area(geom) AS area_m2
FROM mp2025_landuse  -- EPSG:3414, metres
WHERE planning_area = 'BISHAN'
  AND lu_desc IN ('RESIDENTIAL')
  AND gpr_numeric > 3  -- non-numeric GPR (LND/EVA/SDP) excluded
ORDER BY objectid
